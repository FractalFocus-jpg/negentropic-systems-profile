"not_certified"}};
  const audit=auditPolicy(auditByScale,policy.thresholds,K,family);
  if(audit.pass) return {...policy,audit};
  return {...policy,certified:false,thresholds:Object.fromEntries(SCALES.map(s=>[s,Array(K).fill(1.0)])),audit};
}
function route(rows,thresholds,method,witnessCount){
  const eligible=method==="review_only"?rows.map(()=>false):thresholdEligible(rows,thresholds);
  const execute=eligible.slice();
  const remaining=[]; for(let i=0;i<rows.length;i++) if(rows[i].hard&&!execute[i])remaining.push(i);
  remaining.sort((a,b)=>rows[b].rank-rows[a].rank||a-b);
  const review=new Set(remaining.slice(0,Math.floor(REVIEW_FRACTION*rows.length)));
  let exec=0,wrong=0,correct=0,invalid=0,reviewed=review.size; const cn={},ck={};
  for(let i=0;i<rows.length;i++) if(execute[i]&&!review.has(i)){
    const r=rows[i];exec++; if(!r.hard)invalid++; cn[r.y]=(cn[r.y]||0)+1; if(r.pred!==r.y){wrong++;ck[r.y]=(ck[r.y]||0)+1}else correct++;
  }
  let worst=0,eligibleClasses=0;
  for(const [c,n] of Object.entries(cn)) if(n>=50){eligibleClasses++; worst=Math.max(worst,(ck[c]||0)/n);}
  const utility=correct-ERROR_PENALTY*wrong+(1-REVIEW_COST)*reviewed-WITNESS_COST*witnessCount;
  return {n:rows.length,executed:exec,wrong,correct,reviewed,invalid,harm:exec?wrong/exec:0,coverage:exec/rows.length,useful:(correct+reviewed)/rows.length,utility_per_1000:1000*utility/rows.length,worst_class_harm:worst,eligible_classes:eligibleClasses};
}
function s308Thresholds(family,dataset,regime,scale,K){
  const key=`${family}|${dataset}|${regime}|${scale}`; const src=S308_ATLAS[key]||{}; return Array.from({length:K},(_,i)=>src[String(i)]??0.99);
}
function aggregateRun(method,cells){
  const byDataset={}; for(const d of Object.keys(DATASETS))byDataset[d]=0;
  let risk=true,cls=true,useful=true,utility=true,authority=true,reviewNonreg=true,strictReview=0;
  for(const c of cells){
    if(c.metrics.harm>TARGET_AGG+1e-12)risk=false;
    if(c.metrics.worst_class_harm>TARGET_CLASS+1e-12)cls=false;
    if(c.metrics.useful<0.10-1e-12)useful=false;
    if(c.metrics.utility_per_1000<=0)utility=false;
    if(c.metrics.invalid!==0)authority=false;
    if(c.metrics.coverage>=0.01)byDataset[c.dataset]++;
    if(c.metrics.utility_per_1000+1e-9<c.review_utility)reviewNonreg=false;
    if(c.metrics.utility_per_1000>c.review_utility+1e-9)strictReview++;
  }
  const autonomy=Object.fromEntries(Object.entries(byDataset).map(([d,n])=>[d,n]));
  const autonomyPass=Object.values(byDataset).every(n=>n>=12);
  const complete=risk&&cls&&useful&&utility&&authority&&autonomyPass&&reviewNonreg&&strictReview>=18;
  return {method,complete,risk,subgroup:cls,useful,utility,authority,autonomy,autonomy_pass:autonomyPass,review_nonregression:reviewNonreg,strict_review_advantage_cells:strictReview,worst_aggregate_harm:Math.max(...cells.map(c=>c.metrics.harm)),worst_class_harm:Math.max(...cells.map(c=>c.metrics.worst_class_harm)),min_useful:Math.min(...cells.map(c=>c.metrics.useful)),min_utility:Math.min(...cells.map(c=>c.metrics.utility_per_1000)),total_utility:sum(cells.map(c=>c.metrics.utility_per_1000))};
}
