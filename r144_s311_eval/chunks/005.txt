  const result={stable_id:STABLE_ID,title:TITLE,mechanism:"action-conditional Learn-Then-Test / conformal decision calibration with intersection-union release and independent audit veto",external_provenance:["Angelopoulos et al., Learn then Test (2021/2026 revision)","Lekeufack et al., Conformal Decision Theory (2023)","Zhu et al., Action-Conditional Conformal Risk-Averse Decision Making (ICML 2026)"],hidden_bank:{units_per_family:unitsPerFamily,families:FAMILIES,seed_control:"Railway runtime-generated 256-bit secret",secret_disclosed_after_close:secret},family_summary:familySummary,promotion,comparative_promotion:comparativePromotion,disposition:promotion?"INDEPENDENT_HOST_SYNTHETIC_RELIABILITY_PASS_PROCEDURAL_INDEPENDENCE_ONLY":"INDEPENDENT_HOST_SYNTHETIC_RELIABILITY_FAIL",runs,elapsed_ms:Date.now()-started,limitations:["Same research operator specified proposer and evaluator source; this is procedural and host separation, not unaffiliated institutional independence.","Synthetic score-generation workload, not real-world domain validation.","Reliability promotion and comparative superiority are reported separately.","No cybersecurity or quantum-cryptographic superiority inference."]};

async function handle(url){
  const p=new URL(url,"http://localhost").pathname;
  if(p==="/"||p==="/commit") return {status:200,body:{stable_id:STABLE_ID,title:TITLE,phase:state.phase,commitment,source_sha256:sourceHash,protocol_sha256:protocolHash,created_at:state.createdAt}};
  if(p==="/run"){
    if(!state.result){ state.phase="RUNNING"; state.result=await evaluate(masterSecret); state.phase="CLOSED"; }
    return {status:200,body:{phase:state.phase,commitment,result_sha256:state.result.result_sha256,promotion:state.result.promotion,family_summary:state.result.family_summary,elapsed_ms:state.result.elapsed_ms}};
  }
  if(p==="/result") return state.result?{status:200,body:{...state.result,commitment,source_sha256:sourceHash,protocol_sha256:protocolHash}}:{status:409,body:{phase:state.phase,error:"evaluation not closed"}};
  if(p==="/replay"){
    if(!state.result)return {status:409,body:{error:"run first"}};
    const r=await evaluate(masterSecret); state.replay={match:r.result_sha256===state.result.result_sha256,replay_sha256:r.result_sha256,original_sha256:state.result.result_sha256,elapsed_ms:r.elapsed_ms}; return {status:200,body:state.replay};
  }
  return {status:404,body:{error:"not found"}};
}
